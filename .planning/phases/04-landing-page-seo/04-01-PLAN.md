```typescript
// If not logged in AND not explicitly in app mode → show landing page

  return (
    <LandingPage
      onGetStarted={() => setShowAuthModal(true)}  // opens sign-up modal
      onTryFree={() => setShowApp(true)}            // goes directly to tool UI
    />
  );
}
// ... rest of existing app render
```
