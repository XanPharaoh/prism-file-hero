**Update AccountPanel.tsx — replace placeholder alert with real checkout:**

```typescript
const handleUpgrade = async (plan: 'pro' | 'business') => {
  if (!session) return;
  setIsUpgrading(true);
  try {
    const res = await fetch(`${WORKER_BASE_URL}/api/checkout`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${session.access_token}`,
      },
      body: JSON.stringify({ plan, userEmail: session.user.email }),
    });
    const data = await res.json();
    if (data.url) {
      window.location.href = data.url;  // Redirect to Stripe Checkout
    } else {
      throw new Error(data.error || 'Failed to create checkout session');
    }
  } catch (e: any) {
    setError(e.message);
  } finally {
    setIsUpgrading(false);
  }
};
```
