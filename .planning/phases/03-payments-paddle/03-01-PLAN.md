export async function updateUserPlan(
  env: Env,
  userId: string,
  plan: 'free' | 'pro' | 'business',
  stripeCustomerId?: string,
  stripeSubscriptionId?: string,
  status?: string
): Promise<void> {
  await fetch(
    `${env.SUPABASE_URL}/rest/v1/user_plans?user_id=eq.${userId}`,
    {
      method: 'PATCH',
      headers: {
        apikey: env.SUPABASE_SERVICE_ROLE_KEY,
        Authorization: `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        plan,
        ...(stripeCustomerId ? { stripe_customer_id: stripeCustomerId } : {}),
        ...(stripeSubscriptionId ? { stripe_subscription_id: stripeSubscriptionId } : {}),
        ...(status ? { subscription_status: status } : {}),
        updated_at: new Date().toISOString(),
      }),
    }
  );
}

export async function getUserIdByStripeCustomer(env: Env, customerId: string): Promise<string | null> {
  const res = await fetch(
    `${env.SUPABASE_URL}/rest/v1/user_plans?stripe_customer_id=eq.${customerId}&select=user_id&limit=1`,
    {
      headers: {
        apikey: env.SUPABASE_SERVICE_ROLE_KEY,
        Authorization: `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}`,
      },
    }
  );
  const rows = await res.json() as any[];
  return rows[0]?.user_id ?? null;
}

import { Env, updateUserPlan, getUserIdByStripeCustomer } from './db';

export async function createCheckoutSession(
  env: Env,
  userId: string,
  userEmail: string,
  priceId: string,
  successUrl: string,
  cancelUrl: string
): Promise<string> {
  const params = new URLSearchParams({
    'mode': 'subscription',
    'customer_email': userEmail,
    'line_items[0][price]': priceId,
    'line_items[0][quantity]': '1',
    'success_url': successUrl,
    'cancel_url': cancelUrl,
    'metadata[user_id]': userId,
    'subscription_data[metadata][user_id]': userId,
    'allow_promotion_codes': 'true',
  });

  const res = await fetch('https://api.stripe.com/v1/checkout/sessions', {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${env.STRIPE_SECRET_KEY}`,
      'Content-Type': 'application/x-www-form-urlencoded',
    },
    body: params.toString(),
  });

  const session = await res.json() as any;
  if (!res.ok) throw new Error(session.error?.message ?? 'Stripe error');
  return session.url as string;
}

// Verify Stripe webhook signature using Web Crypto
export async function verifyStripeSignature(
  payload: string,
  sigHeader: string,
  secret: string
): Promise<boolean> {
  try {
    const parts = sigHeader.split(',').reduce((acc, part) => {
      const [k, v] = part.split('=');
      acc[k] = v;
      return acc;
    }, {} as Record<string, string>);

    const timestamp = parts['t'];
    const sig = parts['v1'];
    if (!timestamp || !sig) return false;

    const signedPayload = `${timestamp}.${payload}`;
    const encoder = new TextEncoder();
    const key = await crypto.subtle.importKey(
      'raw', encoder.encode(secret),
      { name: 'HMAC', hash: 'SHA-256' },
      false, ['sign']
    );
    const signatureBytes = await crypto.subtle.sign('HMAC', key, encoder.encode(signedPayload));
    const computed = Array.from(new Uint8Array(signatureBytes))
      .map(b => b.toString(16).padStart(2, '0')).join('');

    return computed === sig;
  } catch {
    return false;
  }
}

// Map Stripe price ID to plan name
export function planFromPriceId(priceId: string, env: Env): 'pro' | 'business' {
  if (priceId === env.STRIPE_PRO_PRICE_ID) return 'pro';
  if (priceId === env.STRIPE_BUSINESS_PRICE_ID) return 'business';
  return 'pro'; // default to pro for unknown price
}

// Handle incoming Stripe webhook events
export async function handleStripeWebhook(
  env: Env,
  event: any
): Promise<void> {
  const { type, data } = event;

  if (type === 'customer.subscription.created' || type === 'customer.subscription.updated') {
    const subscription = data.object;
    const userId = subscription.metadata?.user_id;
    const customerId = subscription.customer;
    const priceId = subscription.items?.data[0]?.price?.id;
    const status = subscription.status; // 'active', 'past_due', etc.

    if (!userId && customerId) {
      // Fallback: look up user by customer ID
      const foundUserId = await getUserIdByStripeCustomer(env, customerId);
      if (!foundUserId) return;
      const plan = status === 'active' ? planFromPriceId(priceId, env) : 'free';
      await updateUserPlan(env, foundUserId, plan, customerId, subscription.id, status);
      return;
    }

    if (userId) {
      const plan = status === 'active' ? planFromPriceId(priceId, env) : 'free';
      await updateUserPlan(env, userId, plan, customerId, subscription.id, status);
    }
  }

  if (type === 'customer.subscription.deleted') {
    const subscription = data.object;
    const userId = subscription.metadata?.user_id;
    const customerId = subscription.customer;

    const targetUserId = userId || await getUserIdByStripeCustomer(env, customerId);
    if (targetUserId) {
      await updateUserPlan(env, targetUserId, 'free', customerId, undefined, 'cancelled');
    }
  }
}

if (url.pathname === '/api/checkout' && request.method === 'POST') {
  // Auth required
  const authHeader = request.headers.get('Authorization');
  if (!authHeader?.startsWith('Bearer ')) {
    return new Response(JSON.stringify({ error: 'Authentication required' }), { status: 401, headers: cors });
  }
  const payload = await verifyJWT(authHeader.slice(7), env.SUPABASE_JWT_SECRET);
  if (!payload) {
    return new Response(JSON.stringify({ error: 'Invalid token' }), { status: 401, headers: cors });
  }

  const body = await request.json() as any;
  const { plan, userEmail } = body;
  const priceId = plan === 'business' ? env.STRIPE_BUSINESS_PRICE_ID : env.STRIPE_PRO_PRICE_ID;
  const origin = request.headers.get('Origin') || env.ALLOWED_ORIGIN;

  try {
    const checkoutUrl = await createCheckoutSession(
      env,
      payload.sub,
      userEmail,
      priceId,
      `${origin}/account?upgraded=true`,
      `${origin}/pricing`,
    );
    return new Response(JSON.stringify({ url: checkoutUrl }), {
      status: 200,
      headers: { ...Object.fromEntries(cors), 'Content-Type': 'application/json' },
    });
  } catch (e: any) {
    return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: cors });
  }
}

if (url.pathname === '/api/webhooks/stripe' && request.method === 'POST') {
  const sigHeader = request.headers.get('stripe-signature') ?? '';
  const rawBody = await request.text();

  const valid = await verifyStripeSignature(rawBody, sigHeader, env.STRIPE_WEBHOOK_SECRET);
  if (!valid) {
    return new Response(JSON.stringify({ error: 'Invalid signature' }), { status: 400 });
  }

  const event = JSON.parse(rawBody);
  await handleStripeWebhook(env, event);

  return new Response(JSON.stringify({ received: true }), {
    status: 200,
    headers: { 'Content-Type': 'application/json' },
  });
}
