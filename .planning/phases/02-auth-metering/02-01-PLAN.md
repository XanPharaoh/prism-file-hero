```sql
-- Enable UUID extension
create extension if not exists "pgcrypto";

-- User plan table (one row per user)
create table if not exists public.user_plans (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null unique references auth.users(id) on delete cascade,
  plan text not null default 'free' check (plan in ('free', 'pro', 'business')),
  stripe_customer_id text,
  stripe_subscription_id text,
  subscription_status text default 'inactive',
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now()
);

-- Usage log (one row per month per user — reset monthly)
create table if not exists public.usage_logs (
  id uuid primary key default gen_random_uuid(),
  user_id uuid not null references auth.users(id) on delete cascade,
  month text not null,  -- format: "2026-03"
  pages_used integer not null default 0,
  created_at timestamptz not null default now(),
  updated_at timestamptz not null default now(),
  unique (user_id, month)
);

-- Auto-create user_plans row on new signup
create or replace function public.handle_new_user()
returns trigger as $$
begin
  insert into public.user_plans (user_id)
  values (new.id)
  on conflict (user_id) do nothing;
  return new;
end;
$$ language plpgsql security definer;

drop trigger if exists on_auth_user_created on auth.users;
create trigger on_auth_user_created
  after insert on auth.users
  for each row execute procedure public.handle_new_user();

-- RLS: users can only read their own data
alter table public.user_plans enable row level security;
alter table public.usage_logs enable row level security;

create policy "Users read own plan"
  on public.user_plans for select using (auth.uid() = user_id);

create policy "Users read own usage"
  on public.usage_logs for select using (auth.uid() = user_id);
-- Service role key bypasses RLS (used in Worker)
```

```typescript
export type Plan = 'free' | 'pro' | 'business';

export const PLAN_LIMITS: Record<Plan, number> = {
  free: 10,       // pages per month
  pro: 500,       // pages per month
  business: -1,   // unlimited (-1 = no limit)
};

export function isWithinLimit(plan: Plan, pagesUsed: number, pagesRequested: number): boolean {
  const limit = PLAN_LIMITS[plan];
  if (limit === -1) return true; // unlimited
  return pagesUsed + pagesRequested <= limit;
}
```

```typescript
export interface Env {
  GROQ_API_KEY: string;
  ALLOWED_ORIGIN: string;
  SUPABASE_URL: string;
  SUPABASE_SERVICE_ROLE_KEY: string;
}

export async function getUserPlan(env: Env, userId: string): Promise<'free' | 'pro' | 'business'> {
  const res = await fetch(
    `${env.SUPABASE_URL}/rest/v1/user_plans?user_id=eq.${userId}&select=plan&limit=1`,
    {
      headers: {
        apikey: env.SUPABASE_SERVICE_ROLE_KEY,
        Authorization: `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}`,
      },
    }
  );
  const rows = await res.json() as any[];
  return (rows[0]?.plan ?? 'free') as 'free' | 'pro' | 'business';
}

export async function getCurrentUsage(env: Env, userId: string, month: string): Promise<number> {
  const res = await fetch(
    `${env.SUPABASE_URL}/rest/v1/usage_logs?user_id=eq.${userId}&month=eq.${month}&select=pages_used&limit=1`,
    {
      headers: {
        apikey: env.SUPABASE_SERVICE_ROLE_KEY,
        Authorization: `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}`,
      },
    }
  );
  const rows = await res.json() as any[];
  return rows[0]?.pages_used ?? 0;
}

export async function incrementUsage(env: Env, userId: string, month: string, pages: number): Promise<void> {
  // Upsert: insert or add to existing count
  await fetch(`${env.SUPABASE_URL}/rest/v1/usage_logs`, {
    method: 'POST',
    headers: {
      apikey: env.SUPABASE_SERVICE_ROLE_KEY,
      Authorization: `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}`,
      'Content-Type': 'application/json',
      Prefer: 'resolution=merge-duplicates',
    },
    body: JSON.stringify({
      user_id: userId,
      month,
      pages_used: pages,
    }),
  });

  // Then increment if it already existed
  await fetch(
    `${env.SUPABASE_URL}/rest/v1/usage_logs?user_id=eq.${userId}&month=eq.${month}`,
    {
      method: 'PATCH',
      headers: {
        apikey: env.SUPABASE_SERVICE_ROLE_KEY,
        Authorization: `Bearer ${env.SUPABASE_SERVICE_ROLE_KEY}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({ pages_used: `pages_used + ${pages}` }),
    }
  );
}
```

```typescript
// Insert before the /api/ai handler:
async function verifyJWT(token: string, secret: string): Promise<{ sub: string } | null> {
  // Cloudflare Workers Web Crypto API
  try {
    const encoder = new TextEncoder();
    const keyData = encoder.encode(secret);
    const key = await crypto.subtle.importKey(
      'raw', keyData, { name: 'HMAC', hash: 'SHA-256' }, false, ['verify']
    );
    const [headerB64, payloadB64, sigB64] = token.split('.');
    const signingInput = `${headerB64}.${payloadB64}`;
    const sig = Uint8Array.from(atob(sigB64.replace(/-/g, '+').replace(/_/g, '/')), c => c.charCodeAt(0));
    const valid = await crypto.subtle.verify('HMAC', key, sig, encoder.encode(signingInput));
    if (!valid) return null;
    const payload = JSON.parse(atob(payloadB64.replace(/-/g, '+').replace(/_/g, '/')));
    if (payload.exp && payload.exp < Date.now() / 1000) return null;
    return payload;
  } catch {
    return null;
  }
}
```
